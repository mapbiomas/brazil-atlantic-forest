/**
 * This script performs a temporal filter analysing a five-year moving window to correct
 * a 3-year classification change. It focuses on refining the classification of specific classes (3, 4, 11, 12, 21, 22, 29, 50).
 * 
 * It improves the temporal consistency of the classification by identifying and correcting specific,
 * short-lived classification changes over the years, when pixels of the land cover classification
 * changes to a new type for just three years before changing back.
 * It assumes this short-lived change is a classification error, not a real change on the ground.
 * 
 * The ‘window5y’ function works in the same way as the 4-year temporal window.
 * The only difference is that it is used three separate times throughout the script.
 * Each time is used a different list of years (anos_1, anos_3, anos_2)
 * that are stepped by -3 but start at a different offset (2021, 2023, 2022),
 * to ensure that the moving window covers every possible 3-year interval in the time series
 * and that each modification made on previous iterations wont affect the others coming after.
 * 
 * It uses as input data output data from script 08-10.
 * The output data from this script is used as an input in script 09-10.
 * 
 */

// Function to correct band names.
var corrIndx  = function (img){
                  var indxNames = img.bandNames();
                  var bandNames = indxNames.map(function(nome){
                          return ee.String(nome).split('_').slice(1).join('_');
                                      });
                                      
                    return img.select(indxNames,bandNames);
                              };
                              
// Get the last three years from the 'anos' list.

var antpen = ee.Number(anos.get(n.subtract(3))); // Get the third to last year.

print(ultimo);
print(penult);

// Create first list of years for the 5-year window.
var anos_1 = ee.List.sequence(2021,1988,-3)
                  .map(function(y){
                        return ee.Number(y).int(); });
                        
// Function to apply a 5-year temporal moving window filter.
var window5y = function(img, classe){
                    
 // Create initial image with bands for the last three years.
 var class_final =                      img.select(ee.String('classification_').cat(ultimo)).rename(ee.String('00_classification_').cat(ultimo));
     class_final = class_final.addBands(img.select(ee.String('classification_').cat(penult)).rename(ee.String('00_classification_').cat(penult)));
     class_final = class_final.addBands(img.select(ee.String('classification_').cat(antpen)).rename(ee.String('00_classification_').cat(antpen)));
                      
                  // Create an image collection for each year in the 5-year window.
                  var classWind = ee.ImageCollection(anos_1.map(function(ano){
                    ano = ee.Number(ano);
                    var anoStr = ano.format();  
                    var nomeBanda = ee.String('classification_').cat(anoStr);
                    
                    // Select the classification for the current year.
                    var class_ano = img.select(ee.String('classification_').cat(anoStr));
                    
                    // Create a mask based on the condition that the current year's class is equal to the input class,
                    // and the previous and next years are also equal (eq) to the input class, while the two years before
                    // and after are not equal (neq) to the input class.
                    var mask_5 = img.select(ee.String('classification_').cat(ano.add(1).format()))     .neq(classe)
                            .and(img.select(ee.String('classification_').cat(anoStr))                   .eq(classe))
                            .and(img.select(ee.String('classification_').cat(ano.subtract(1).format())) .eq(classe))
                            .and(img.select(ee.String('classification_').cat(ano.subtract(2).format())) .eq(classe))
                            .and(img.select(ee.String('classification_').cat(ano.subtract(3).format())).neq(classe));
                            
                    // Remap the classification for the previous year and apply the mask.
                    mask_5 = img.select(ee.String('classification_').cat(ano.subtract(3).format()))
                                .remap([3,4,11,12,21,22,29,50],
                                       [3,4,11,12,21,22,29,50]).updateMask(mask_5);
                    
                    // Blend the original classification with the masked classification.
                    var class_corr = class_ano.blend(mask_5.rename(ee.String('classification_').cat(anoStr)));
                    // Blend the original classification for the previous year with the masked classification.
                    var class_corr2 = img.select(ee.String('classification_').cat(ano.subtract(1).format()))
                                         .blend(mask_5.rename(ee.String('classification_').cat(ano.subtract(1).format())));
                    // Blend the original classification for the previous year with the masked classification.
                    var class_corr3 = img.select(ee.String('classification_').cat(ano.subtract(2).format()))
                                         .blend(mask_5.rename(ee.String('classification_').cat(ano.subtract(2).format())));
 
                    // Return the blended images for the current and surrounding years.
                    return class_corr.addBands(class_corr2).addBands(class_corr3);
                  })).toBands();
                  
    // Add the initial image bands and the processed image collection bands.
    class_final = class_final.addBands(classWind).addBands(img.select('classification_1985').rename(ee.String('00_classification_1985')));

// Correct band names and return the result.
var corrigidaFinal = corrIndx(class_final);
        // print('corrigidaFinal',corrigidaFinal);
                return corrigidaFinal;
};

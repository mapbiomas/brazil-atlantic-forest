/**
 * This script uses superpixel clustering that is based on SNIC (Simple Non-Iterative Clustering)
 * and segments an image into small compact regions with homogeneity.
 * 
 * Creates a segmented mosaic based on SNIC for each year (1985 to 2024) from LandSat images.
 * 
 * The following bands are added: NDFI amplitude, green texture, longitude, and latitude.
 * 
 * This new image will be used as an input on another script 03-10.
 */

// Define the area of interest.
var limite_mos = /* color: #d63000 */ee.Geometry.Polygon(
        [[[-37.962054750850626, -2.333004463449096],
          [-59.143695375850626, -18.442239000785516],
          [-58.528461000850626, -30.641824948326004],
          [-52.991351625850626, -34.709232071195586],
          [-37.171039125850626, -20.431274632006023],
          [-30.509630101854242, -3.5739518333226985],
          [-35.11290158622924, -2.2683059430342354]]]);

// Visualization parameters for NDVI amplitude.
var imageVisParam = {"opacity":1,"bands":["ndvi_amp"],"min":700,"max":10000,
                     "palette":["137b01","03c40b","34ff04","f9ff00","ffa802","ff2b06","b41503"]};

// Array of years to process.
var anos = [1985,1986,1987,1988,1989,1990,1991,1992,1993,1994,
            1995,1996,1997,1998,1999,2000,2001,2002,2003,2004,
            2005,2006,2007,2008,2009,2010,2011,2012,2013,2014,
            2015,2016,2017,2018,2019,2020,2021,2022,2023,2024];

// Visualization parameters for median composite.
var visParMedian2 = {bands: ['swir1_median', 'nir_median', 'red_median'],gain: [0.08, 0.06, 0.2],gamma: 0.85};

// Define the biome to process.
var bioma = "MATAATLANTICA";

// Asset path for LandSat mosaics.
var asset_mosaicos = 'projects/nexgenmap/MapBiomas2/LANDSAT/BRAZIL/mosaics-2';

// Loop through each year in 'anos'.
for (var i_ano=0;i_ano<anos.length; i_ano++){
  // Get the current year.
  var ano = anos[i_ano];

  // Define the years and LandSat assets for the current year. l5 = Landsat 5, l7 = Landsat 7, l8 = Landsat 8.
  // For each year, the current year and the two immediately preceding years are analyzed.
  // This rule is different for the first two years of the series.
  if      (ano == 1985) { // In 1985, the year itself and the two subsequent years are analyzed.
    var ano1 = 1985; var ano2 = 1986; var ano3 = 1987;
    var asset1 = 'l5'; var asset2 = 'l5'; var asset3 = 'l5';
  }
  else if (ano == 1986) { // In 1986, the year itself, the previous year and the following year are analyzed.
    var ano1 = 1986; var ano2 = 1985; var ano3 = 1987;
    var asset1 = 'l5'; var asset2 = 'l5'; var asset3 = 'l5';
  }
  else if (ano == 2000) {
    var ano1 = 2000; var ano2 = 1999; var ano3 = 1998;
    var asset1 = 'l7'; var asset2 = 'l5'; var asset3 = 'l5';
  }
  else if (ano == 2001)   {
    var ano1 = 2001; var ano2 = 2000; var ano3 = 1999;
    var asset1 = 'l7'; var asset2 = 'l7'; var asset3 = 'l5';
  }
  else if (ano == 2002) {
    var ano1 = 2002; var ano2 = 2001; var ano3 = 2000;
    var asset1 = 'l7'; var asset2 = 'l7'; var asset3 = 'l7';
  }
  else if (ano == 2003) {
    var ano1 = 2003; var ano2 = 2002; var ano3 = 2001;
     var asset1 = 'l5'; var asset2 = 'l7'; var asset3 = 'l7';
  }
  else if (ano == 2004) {
    var ano1 = 2004; var ano2 = 2003; var ano3 = 2002;
     var asset1 = 'l5'; var asset2 = 'l5'; var asset3 = 'l7';
  }
  else if (ano == 2012) {
    var ano1 = 2012; var ano2 = 2011; var ano3 = 2010;
    var asset1 = 'l7'; var asset2 = 'l5'; var asset3 = 'l5';
  }
  else if (ano == 2013) {
    var ano1 = 2013; var ano2 = 2012; var ano3 = 2011;
    var asset1 = 'l8'; var asset2 = 'l7'; var asset3 = 'l5';
  }
  else if (ano == 2014) {
    var ano1 = 2014; var ano2 = 2013; var ano3 = 2012;
    var asset1 = 'l8'; var asset2 = 'l8'; var asset3 = 'l7';
  }
  else if (ano == 2015) {
    var ano1 = 2015; var ano2 = 2014; var ano3 = 2013;
    var asset1 = 'l8'; var asset2 = 'l8'; var asset3 = 'l8';
  }
  else if (ano == 2016) {
    var ano1 = 2016; var ano2 = 2015; var ano3 = 2014;
    var asset1 = 'l8'; var asset2 = 'l8'; var asset3 = 'l8';
  }
  else if (ano == 2017) {
    var ano1 = 2017; var ano2 = 2016; var ano3 = 2015;
    var asset1 = 'l8'; var asset2 = 'l8'; var asset3 = 'l8';
  }
  else if (ano == 2018) {
    var ano1 = 2018; var ano2 = 2017; var ano3 = 2016;
    var asset1 = 'l8'; var asset2 = 'l8'; var asset3 = 'l8';
  }
  else if (ano == 2019) {
    var ano1 = 2019; var ano2 = 2018; var ano3 = 2017;
    var asset1 = 'l8'; var asset2 = 'l8'; var asset3 = 'l8';
  }
  else if (ano == 2020) {
    var ano1 = 2020; var ano2 = 2019; var ano3 = 2018;
    var asset1 = 'l8'; var asset2 = 'l8'; var asset3 = 'l8';
  }
  else if (ano == 2021) {
    var ano1 = 2021; var ano2 = 2020; var ano3 = 2019;
    var asset1 = 'l8'; var asset2 = 'l8'; var asset3 = 'l8';
  }
  else if (ano == 2022) {
    var ano1 = 2022; var ano2 = 2021; var ano3 = 2020;
    var asset1 = 'l8'; var asset2 = 'l8'; var asset3 = 'l8';
  }
  else if (ano == 2023) {
    var ano1 = 2023; var ano2 = 2022; var ano3 = 2021;
    var asset1 = 'l8'; var asset2 = 'l8'; var asset3 = 'l8';
  }
  else if (ano == 2024) {
    var ano1 = 2024; var ano2 = 2023; var ano3 = 2022;
    var asset1 = 'l8'; var asset2 = 'l8'; var asset3 = 'l8';
  }
  else { // For the years between 1987 and 1999, and between 2005 and 2011, the year itself and the two previous years are analyzed, with Landsat 5.
    var ano1 = ano; var ano2 = ( ano - 1); var ano3 = ( ano - 2);
    var asset1 = 'l5'; var asset2 = 'l5'; var asset3 = 'l5';
  }

  // Create a mosaic for the current year (ano1).
  var mosaicoTotal = ee.ImageCollection(asset_mosaicos)
                       .filterMetadata('biome', 'equals', bioma)
                       .filterMetadata('year', 'equals', ano1)
                       .filter(ee.Filter.eq('satellite', asset1))
                       .mosaic();
  
  // Apply the Segmentation SNIC algorithm to the created mosaic. Superpixel clustering based on SNIC (Simple Non-Iterative Clustering).
  var snic = ee.Algorithms.Image.Segmentation.SNIC({
    image: mosaicoTotal.select(['blue_median','green_median','red_median','nir_median',
    'swir1_median','ndfi_median','green_median_texture']),
    size: 20,
    compactness: 0.4,
    connectivity: 8,
  });
  
  // Create mosaics for the previous two years (ano2 and ano3).
  var mosaico1ano_antes = ee.ImageCollection(asset_mosaicos)
                    .filterMetadata('biome', 'equals', bioma)
                    .filterMetadata('year', 'equals', ano2)
                    .filter(ee.Filter.eq('satellite', asset2))
                    .mosaic();
  var mosaico2anos_antes = ee.ImageCollection(asset_mosaicos)
                    .filterMetadata('biome', 'equals', bioma)
                    .filterMetadata('year', 'equals', ano3)
                    .filter(ee.Filter.eq('satellite', asset3))
                    .mosaic();

  // Calculate the minimum and maximum NDFI values for the current and previous years (ano1, ano2, ano3).
  // Minimum NDFI
  var min3anos = ee.ImageCollection.fromImages([mosaicoTotal.select('ndfi_median_wet'),
                                                mosaico1ano_antes.select('ndfi_median_wet'),
                                                mosaico2anos_antes.select('ndfi_median_wet')]).min();
  // Maximum NDFI
  var max3anos = ee.ImageCollection.fromImages([mosaicoTotal.select('ndfi_median_wet'),
                                                mosaico1ano_antes.select('ndfi_median_wet'),
                                                mosaico2anos_antes.select('ndfi_median_wet')]).max();
  
  // Calculate the amplitude of NDFI over three years. (Maximum NDFI - Minimum NDFI)
  var amp3anos_ndfi = max3anos.subtract(min3anos).rename('amp_ndfi_3anos');

  // Add the NDFI amplitude to the segmented SNIC image.
  snic = snic.addBands(amp3anos_ndfi);

  // Calculate the longitude and latitude of each pixel.
  var ll = ee.Image.pixelLonLat().clip(limite_mos);                 

  var long = ll.select('longitude').add(34.8).multiply(-1).multiply(1000).toInt16();
  var lati = ll.select('latitude').add(5).multiply(-1).multiply(1000).toInt16();

  // Add the longitude and latitude to the segmented SNIC image.
  snic = snic.addBands(long.rename('longitude'));
  snic = snic.addBands(lati.rename('latitude' ));

  // Add green texture and NDFI median bands.
  var green_text = snic.select(["green_median_texture_mean","ndfi_median_mean"],
                               ["clusters_green_text_"+ano1 , "clusters_ndfi_median_"+ano1])
                               .toInt8();

  // Builds a multi-band image (`mosaicoMA`) by iteratively adding bands from the years list.
  // Create a mosaic of the segmented SNIC images for all years.
  // The first iteration handles the initial image creation, subsequent iterations add bands.
  if (i_ano == 0){ // It checks if it's the first iteration (i_ano == 0).
    // If it's the first iteration, it initializes `mosaicoMA` with bands from the SNIC segmentation,
    // NDFI amplitude, green texture, longitude, and latitude.
    var mosaicoMA = snic.select(["clusters"],["clusters_"+ano1])
                        .addBands(amp3anos_ndfi.rename("amp_ndfi_3anos_"+ano1 ))
                        .addBands(green_text)
                        .addBands(long.rename('longitude'))
                        .addBands(lati.rename('latitude' ));
                  }
  else {
    // If it's not the first iteration, it adds bands from the current year's SNIC segmentation,
    // NDFI amplitude, and green texture to the existing `mosaicoMA` image.
    mosaicoMA = mosaicoMA.addBands(snic.select(["clusters"],["clusters_"+ano1])
                         .addBands(amp3anos_ndfi.rename("amp_ndfi_3anos_"+ano1 ))
                         .addBands(green_text));
       }
}

// Print the mosaic.
print(mosaicoMA);

// Export the mosaic to an asset.
Export.image.toAsset({
  "image": mosaicoMA,
  "description": 'mosaicoMA_clusters_1985_2024_asset_salvo',
  "assetId": 'projects/mapbiomas-workspace/COLECAO_10/MA_clusters_1985_2024_asset_salvo',
  "scale": 30,
  "pyramidingPolicy": {
      '.default': 'mode'
  },
  "maxPixels": 1e13,
  "region": limite_mos
});

/**
 * This script adds Rock Outcrop as a new class over Grassland areas.
 * 
 * It analyses spectral indices (NDVI) derived from Landsat data.
 * It calculates some NDVI metrics (amplitude, standard deviation, median, median for dry and wet seasons,
 * minimum, and maximum), and creates masks based on thresholds for these metrics.
 * It combines the NDVI masks, creating a binary mask (afloramentoAno) where 1 indicates a potential rock outcrop.
 * The rock outcrop mask is added to the original land cover classification (imgCol),
 * adding the Rock Outcrop class (29) over class Grassland (12). Then blends the original and updated classifications.
 * 
 * It uses as input data output data from script 06-20.
 * The output data from this script is used as an input in script 07-10.
 * 
 */

// Define the geometry for the Atlantic Forest region.
var limite_MA = /* color: #d63000 */ee.Geometry.Polygon(
        [[[-48.593359954293625, -30.678347823900353],
          [-47.275000579293625, -25.525376684152373],
          [-40.595313079293625, -23.284530667538736],
          [-33.915625579293625, -6.580343714417967],
          [-35.453711516793625, -4.217995607905081],
          [-44.198828704293625, -17.856203449528717],
          [-50.483008391793625, -17.52126295946964],
          [-55.712500579293625, -21.74193426005608],
          [-55.492774016793625, -29.72888025446976]]]);

// Define input and output version numbers.
var vesion_in = '6';
var versao_out = '40';
// Define the description of the process.
var descricao = 'Adiciona afloramento rochoso';
// Define the collection id.
var col = 10.0;
// Define input and output prefixes for asset naming.
var prefixo_in  = 'MA_col'+col+'_p06b_v';
var prefixo_out = 'MA_col'+col+'_p06c_v'; 
// Define the output directory.
var dirout = 'projects/mapbiomas-brazil/assets/LAND-COVER/COLLECTION-'+col+'/GENERAL/classification-mat-ft/';

// Load the collection 10 image.
var imgCol =  ee.Image(dirout+prefixo_in+vesion_in);

// Import the palettes module.
var palettes = require('users/mapbiomas/modules:Palettes.js');

// Define visualization parameters.
var vis = {
    'min': 0,
    'max': 69,
    'palette': palettes.get('classification9')
};
var vis2 = {
    'bands': 'classification_2020',
    'min': 0,
    'max': 69,
    'palette': palettes.get('classification9')
};
var visParMedian2 = {bands: ['swir1_median_wet', 'nir_median', 'red_median'],gain: [0.08, 0.06, 0.2],gamma: 0.85};

// Define years to process.
var anos = [1985,1986,1987,1988,1989,1990,1991,1992,1993,1994,
            1995,1996,1997,1998,1999,2000,2001,2002,2003,2004,
            2005,2006,2007,2008,2009,2010,2011,2012,2013,2014,
            2015,2016,2017,2018,2019,2020,2021,2022,2023,2024];

// Initialize an image for accumulating rock outcrop masks.
var mascara_acumulada = ee.Image(1);

// Loop through each year.
for (var i_ano=0;i_ano<anos.length; i_ano++){
  var ano = anos[i_ano];

  // Import the module for accessing monthly mosaics.
  var mos = require('users/marcosrosaUSP/MapBiomas_col10_MA:Mata_Atlantica_LANDSAT/passo01/old_mosaicos_mensais_do_Google_v3');
  // Get the mosaic for the current year and region.
  var mosaicoTotal = mos.getMosaic(ano,limite_MA);
  // Add a band representing the year.
  var img_ano = ee.Image(ano);
  mosaicoTotal = mosaicoTotal.addBands(img_ano.rename('ano'));

  // Define bands to use for rock outcrop detection.
  var bandNames = ee.List([
  'ndvi_amp','ndvi_median','ndvi_median_dry','ndvi_median_wet','ndvi_stdDev','ndvi_min','ndvi_max'
      ]); 
  // Filter and mosaic the image collection.
  var mosaicoNDVI = ee.ImageCollection(mosaicoTotal)
                        .filterMetadata('year', 'equals', ano)
                        .filterBounds(limite_MA)
                        .mosaic()
                        .select(bandNames);

  // Define NDVI thresholds for rock outcrop detection.
  var mask_amp    = mosaicoNDVI.select('ndvi_amp')       .gte(20)  .and(mosaicoNDVI.select('ndvi_amp')       .lte(200));
  var mask_stdDev = mosaicoNDVI.select('ndvi_stdDev')    .gte(0)   .and(mosaicoNDVI.select('ndvi_stdDev')    .lte(50));
  var mask_median = mosaicoNDVI.select('ndvi_median')    .gte(1200).and(mosaicoNDVI.select('ndvi_median')    .lte(1650));
  var mask_dry    = mosaicoNDVI.select('ndvi_median_dry').gte(1200).and(mosaicoNDVI.select('ndvi_median_dry').lte(1650));
  var mask_wet    = mosaicoNDVI.select('ndvi_median_wet').gte(1200).and(mosaicoNDVI.select('ndvi_median_wet').lte(1650));
  var mask_min    = mosaicoNDVI.select('ndvi_min')       .gte(1200).and(mosaicoNDVI.select('ndvi_min')       .lte(1400));
  var mask_max    = mosaicoNDVI.select('ndvi_max')       .gte(1300).and(mosaicoNDVI.select('ndvi_max')       .lte(1500));
             
  // Combine all masks using 'AND' operator.
  var afloramentoAno = mask_amp.and(mask_median).and(mask_dry).and(mask_wet).and(mask_min).and(mask_max);
  afloramentoAno = afloramentoAno.updateMask(afloramentoAno);
  // Remap values to represent rock outcrop.
  afloramentoAno = afloramentoAno.and(afloramentoAno).remap([1],[100]);
  // Add rock outcrop to the classification.
  var class_out_aflora_ano = imgCol.select('classification_'+ano).add(afloramentoAno);  
  // Blend the original and updated classifications.
  // Remaps values 112, 100 + 12 (grassland), to 29 (rock outcrop).
  var class_final = imgCol.select('classification_'+ano).blend(class_out_aflora_ano.remap([112],[29]).rename('classification_'+ano)
); 

  // Accumulate results into a single image.
  if (i_ano == 0){ // first year
    var image = class_final;
  } else { // other years
    image = image.addBands(class_final); 
  }
}

// Add Collection 9 classification to the map.
var colecao = ee.Image('projects/mapbiomas-public/assets/brazil/lulc/collection9/mapbiomas_collection90_integration_v1');
Map.addLayer(colecao.select('classification_2023'), vis, 'LULC Col 9 (LandSat) '+ anos, false);

// Add layers to the map for visualization.
Map.addLayer(mosaicoTotal  , visParMedian2, 'Img_Year_'+ano, true);
Map.addLayer(mosaicoNDVI   , {}, 'NDVI'+ano     , false);
Map.addLayer(mask_amp      , {}, 'mask_amp'     , false);
Map.addLayer(mask_stdDev   , {}, 'mask_stdDev'  , false);
Map.addLayer(mask_median   , {}, 'mask_median'  , false);
Map.addLayer(mask_dry      , {}, 'mask_dry'     , false);
Map.addLayer(mask_wet      , {}, 'mask_wet'     , false);
Map.addLayer(mask_min      , {}, 'mask_min'     , false);
Map.addLayer(mask_max      , {}, 'mask_max'     , false);
Map.addLayer(afloramentoAno, {palette: ['#800000']}, 'Afloramento Rochoso');  

// Add the original and filtered images to the map
Map.addLayer(image, vis2, 'imgCol', false);

// Add the classifications for 2020 to the map.
//Map.addLayer(image.select('classification_2020'), vis, 'imgCol 2020', true);
//Map.addLayer(class_final.select('classification_2024'), vis, 'imgCol 2020', true);

//Map.setCenter(-46.16246053, -21.86496901, 14)

// Set the metadata for the final classification image
image = image
.set('territory', 'MATAATLANTICA')
.set('biome', 'MATAATLANTICA')
.set('source', 'arcplan')
.set('version', versao_out)
.set('year', versao_out)
.set('collection_id', col)
.set('description', descricao);

// Export the final classification image to an asset.
Export.image.toAsset({
    "image": image.toInt8(),
    'description': prefixo_out+versao_out,
    'assetId': dirout+prefixo_out+versao_out,
    "scale": 30,
    "pyramidingPolicy": {
        '.default': 'mode'
    },
    "maxPixels": 1e13,
    "region": limite_MA
});
